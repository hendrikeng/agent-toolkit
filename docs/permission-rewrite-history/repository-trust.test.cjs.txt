const test = require('node:test')
const assert = require('node:assert/strict')
const fs = require('node:fs')
const path = require('node:path')
const { identity, trustedRepositories, isTrusted, updateStore } = require('./repository-trust.cjs')

test('trust binds an exact checkout and Git identity, not parent paths, clones or replacements', () => {
  const scratch = process.env.AGENT_TOOLKIT_SCRATCH_ROOT || path.join(require('node:os').homedir(), 'Code/.agent-toolkit-scratch')
  fs.mkdirSync(scratch, { recursive: true, mode: 0o700 })
  assert.equal(fs.lstatSync(scratch).isSymbolicLink(), false)
  const root = fs.realpathSync(fs.mkdtempSync(path.join(scratch, 'repository-trust-')))
  const store = path.join(root, 'agent/repository-trust.json')
  const repo = path.join(root, 'Code/project')
  const other = path.join(root, 'Code/third-party')
  // Filesystem identities are the subject of this fixture, not Git history.
  for (const directory of [repo, other]) fs.mkdirSync(path.join(directory, '.git/objects'), { recursive: true })
  fs.mkdirSync(path.join(repo, 'src'))
  assert.deepEqual(trustedRepositories(store), [])
  assert.throws(() => updateStore(store, 'approve', [path.dirname(repo)]))
  updateStore(store, 'approve', [repo])
  const approved = trustedRepositories(store)
  assert.equal(approved.length, 1)
  assert.equal(isTrusted(repo, approved), true)
  assert.equal(isTrusted(path.join(repo, 'src'), approved), true)
  assert.equal(isTrusted(other, approved), false)
  assert.equal(isTrusted(path.dirname(repo), approved), false)
  const nested = path.join(repo, 'vendor/checkout')
  fs.mkdirSync(path.join(nested, '.git/objects'), { recursive: true })
  assert.equal(isTrusted(nested, approved), false)
  const dangling = path.join(repo, 'vendor/dangling')
  fs.mkdirSync(dangling)
  fs.symlinkSync(path.join(root, 'missing'), path.join(dangling, '.git'))
  assert.equal(isTrusted(dangling, approved), false, 'do not inherit parent trust through a dangling .git link')
  const worktree = path.join(root, 'worktree'), gitdir = path.join(repo, '.git/worktrees/fixture')
  fs.mkdirSync(worktree); fs.mkdirSync(gitdir, { recursive: true })
  fs.writeFileSync(path.join(worktree, '.git'), `gitdir: ${gitdir}\n`)
  fs.writeFileSync(path.join(gitdir, 'commondir'), '../..\n')
  assert.equal(isTrusted(worktree, approved), false, 'a linked worktree does not inherit general execution trust')
  updateStore(store, 'approve', [worktree])
  assert.equal(isTrusted(worktree, trustedRepositories(store)), true)
  fs.writeFileSync(path.join(gitdir, 'commondir'), path.join(other, '.git') + '\n')
  assert.equal(isTrusted(worktree, approved), false)
  assert.equal(trustedRepositories(store).length, 1, 'changed common-directory identity invalidates the worktree grant')
  fs.symlinkSync(other, path.join(repo, 'outside'))
  assert.equal(isTrusted(path.join(repo, 'outside'), approved), false)
  assert.throws(() => identity(path.join(repo, 'outside')), /symlink/)
  fs.renameSync(path.join(repo, '.git'), path.join(repo, '.git-retained'))
  fs.mkdirSync(path.join(repo, '.git/objects'), { recursive: true })
  assert.deepEqual(trustedRepositories(store), [])
  assert.equal(isTrusted(repo, approved), false)
  updateStore(store, 'approve', [repo])
  assert.equal(trustedRepositories(store).length, 1)
  fs.renameSync(repo, `${repo}-retained`)
  assert.deepEqual(trustedRepositories(store), [])
  updateStore(store, 'revoke', [repo])
  assert.deepEqual(trustedRepositories(store), [])
  fs.chmodSync(store, 0o644)
  assert.throws(() => trustedRepositories(store), /private/)
  const alias = path.join(root, 'dangling-store.json')
  fs.symlinkSync(path.join(root, 'absent.json'), alias)
  assert.throws(() => trustedRepositories(alias), /private/)
  assert.throws(() => updateStore(alias, 'approve', [other]), /private/)
  assert.equal(fs.lstatSync(alias).isSymbolicLink(), true)
  console.log(`Retained trust fixture: ${root}`)
})

test('explicit repository approval covers current and future registered worktrees, not unrelated checkouts', () => {
  const scratch = process.env.AGENT_TOOLKIT_SCRATCH_ROOT || path.join(require('node:os').homedir(), 'Code/.agent-toolkit-scratch')
  fs.mkdirSync(scratch, { recursive: true, mode: 0o700 })
  assert.equal(fs.lstatSync(scratch).isSymbolicLink(), false)
  const root = fs.realpathSync(fs.mkdtempSync(path.join(scratch, 'repository-family-')))
  const repo = path.join(root, 'main'), store = path.join(root, 'agent/trust.json')
  fs.mkdirSync(path.join(repo, '.git/objects'), { recursive: true })
  const register = name => {
    const checkout = path.join(root, 'orca/workspaces', name), admin = path.join(repo, '.git/worktrees', name)
    fs.mkdirSync(checkout, { recursive: true }); fs.mkdirSync(admin, { recursive: true })
    fs.writeFileSync(path.join(checkout, '.git'), `gitdir: ${admin}\n`)
    fs.writeFileSync(path.join(admin, 'commondir'), '../..\n')
    fs.writeFileSync(path.join(admin, 'gitdir'), `${checkout}/.git\n`)
    return { checkout, admin }
  }
  const current = register('current')
  updateStore(store, 'approve', [repo])
  assert.equal(isTrusted(current.checkout, trustedRepositories(store)), false, 'existing exact approvals are not silently broadened')
  updateStore(store, 'approve-repository', [repo])
  const receipt = fs.readFileSync(store)
  assert.equal(isTrusted(current.checkout, trustedRepositories(store)), true)
  assert.throws(() => updateStore(store, 'revoke', [current.checkout]), /main checkout/)
  assert.deepEqual(fs.readFileSync(store), receipt, 'unsupported worktree revocation must not report success or change approvals')
  const future = register('future')
  assert.equal(isTrusted(future.checkout, trustedRepositories(store)), true)
  assert.deepEqual(fs.readFileSync(store), receipt, 'discovery does not rewrite human approvals')
  assert.throws(() => updateStore(store, 'approve-repository', [future.checkout]), /main checkout/)
  const foreign = path.join(root, 'orca/workspaces/unrelated')
  fs.mkdirSync(path.join(foreign, '.git/objects'), { recursive: true })
  assert.equal(isTrusted(foreign, trustedRepositories(store)), false)
  const forged = path.join(root, 'forged')
  fs.mkdirSync(forged)
  fs.writeFileSync(path.join(forged, '.git'), `gitdir: ${current.admin}\n`)
  assert.equal(isTrusted(forged, trustedRepositories(store)), false)
  fs.writeFileSync(path.join(current.admin, 'gitdir'), `${future.checkout}/.git\n`)
  assert.equal(isTrusted(current.checkout, trustedRepositories(store)), false, 'bidirectional registration is required')
  fs.writeFileSync(path.join(current.admin, 'gitdir'), `${current.checkout}/.git\n`)
  fs.writeFileSync(path.join(current.admin, 'commondir'), `${foreign}/.git\n`)
  assert.equal(isTrusted(current.checkout, trustedRepositories(store)), false, 'foreign common-directory identities are rejected')
  updateStore(store, 'approve', [future.checkout])
  const dualApproval = fs.readFileSync(store)
  assert.throws(() => updateStore(store, 'revoke', [future.checkout]), /repository-wide approval/)
  assert.deepEqual(fs.readFileSync(store), dualApproval)
  updateStore(store, 'revoke', [repo, future.checkout])
  assert.equal(isTrusted(future.checkout, trustedRepositories(store)), false)
  updateStore(store, 'approve-repository', [repo])
  fs.renameSync(path.join(repo, '.git'), path.join(repo, '.git-retained'))
  fs.mkdirSync(path.join(repo, '.git/objects'), { recursive: true })
  assert.deepEqual(trustedRepositories(store), [], 'replacing the approved main identity revokes its worktree family')
  console.log(`Retained repository-family fixture: ${root}`)
})
